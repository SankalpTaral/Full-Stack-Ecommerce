import User from "../models/user.js";

export const createUser = async (req, res) => {
  // Step 1 : Validate the request Body
  try {
    const validateUserRequest = validateUser(req.body);

    if (validateUserRequest) {
      return res.status(400).json({
        message: validateUserRequest,
      });
    }
    // if i reach here meaning i passed check

    const { name, email, password  } = req.body;

    // Step 2 : Check if user already exists or not in db

    const existingUser = await User.findOne({ email });

    if (existingUser) {
      return res.status(409).json({
        message: "Email already registered",
      });
    }

    //Step 3: Hash the password

    const hashedPassword = await bcrypt.hash(password, 10);

    // Step4 : Create the User and saves  into the database

    const newUser = await User.create({
      name,
      email,
      password: hashedPassword,
      role : "user" // even if they send admin they should be made user 
    });

    return res.status(201).json({
      message: "Signed up successfully",
    });
  } catch (err) {
    res.status(500).json({
      message: "Internal Server Error",
    });
  }
};

export const loginUser = async (req, res) => {
  try {
    const validateUserRequest = validateUserLogin(req.body);

    if (validateUserRequest) {
      return res.status(400).json({
        message: validateUserRequest,
      });
    }

    // i come here if all  my login body from user is validated

    // Step 2 : Check if user with those credentials exist or no in the db

    const { email, password } = req.body;

    const user = await User.findOne({ email });

    // if user does not exist then return
    if (!user) {
      return res.status(401).json({
        message: "Invalid email or password",
      });
    }

    // Step 3 : Here i come if user exist with that email Check password

    const isPasswordValid = await bcrypt.compare(password, user.password);

    if (!isPasswordValid) {
      return res.status(401).json({
        message: "Invalid email or password",
      });
    }

    // Step 3.1 Generate a jwt token
    const token = jwt.sign({ id: user._id }, process.env.JWT_SECRET, {
      expiresIn: "1h",
    });

    // Step 3.2
    res.cookie("token", token, {
      httpOnly: true,
      secure: process.env.NODE_ENV === "production",
      sameSite: "lax",
      maxAge: 60 * 60 * 1000,
    });

    // Step 4 : return a response that login succesfull

    return res.status(200).json({
      message: "Logged in Successfull",
    });
  } catch (err) {
    return res.status(500).json({
      message: "Internal server error",
    });
  }
};

import mongoose from "mongoose";
import Product from "../models/product.js";
import Cart from "../models/cart.js";

export const addToCart = async (req, res) => {
  try {
    const { productId } = req.body;

    /* Step 1: Validate reqBody

* If user does not send productId in body then it will be treated as undefined -> !undefined !true
* the productId should be mongoDb valid id not a random id
* mongoose.Types.ObjectId.isValid(productId) checks it returns true if its an validId
* so we are reversing it if it returns false  meaning the id is not valid
* then we convert it to true and make it go in if block and return the response 
* product id is invalid 

*/
    if (!productId || !mongoose.Types.ObjectId.isValid(productId)) {
      return res.status(400).json({
        success: false,
        message: "Invalid product ID",
      });
    }

    // Step 2 Check if productId which is sent exist in ur database of product or no

    const product = await Product.findById(productId);

    if (!product) {
      return res.status(404).json({
        success: false,
        message: "Product not found",
      });
    }

    /*
Step 3 — Find the user's cart
Think about what should happen now.
- If the user already has a cart, we should use that cart.
- If the user doesn't have a cart yet, we need to create one.
*/

    const cart = await Cart.findOne({
      userId: req.user._id,
    });

    // if cart contains document meaning user has a cart already
    // cart is null then meaning that user does not have cart

    // Case 1 : If user does not have cart
    // create a cart and add the productId which he sent
    if (!cart) {
      const newCart = await Cart.create({
        userId: req.user._id,
        items: [
          {
            productId: productId,
            quantity: 1,
          },
        ],
      });

      await newCart.populate("items.productId", "name price");

      return res.status(201).json({
        success: true,
        cartOwner: req.user.name,
        message: "Product added to cart successfully",
        cart: newCart,
      });
    }

    // Case 2: If user has a cart already
    // check in his cart that productId which he is trying to add exist or no in his cart
    // if it exist in his cart increase quantity it it does not exist in his cart add it set qty as 1
    else {
      const existingItem = cart.items.find((item) => {
        return item.productId.toString() === productId.toString();
      });
      // exisiting item if found = it returns an matching object in items array  not entire document

      // . find returns the exisiting Item {productId : id and the quantity : }
      // if exist go and increase the qty
      // if existingItem does not find anything means that product does not exist in cart
      // so create a new product in user cart ka items push it id and quantity

      if (existingItem) {
        existingItem.quantity += 1;
      } else {
        cart.items.push({
          productId,
          quantity: 1,
        });
      }

      cart.populate("items.productId", "name price");

      await cart.save();
    }

    return res.status(200).json({
      success: true,
      cartOwner: req.user.name,
      message: "Product added to cart successfully",
      cart: cart,
    });
  } catch (err) {
    return res.status(500).json({
      success: false,
      message: "Internal server error",
    });
  }
};

export const createOrder = async (req, res, next) => {
  try {
    //Step 0
    const validate = BodyOfOrderValidator(req.body);

    if (validate) {
      return res.status(400).json({
        success: false,
        message: validate,
      });
    }
    console.log("After step 1");

    // Step 1
    const userId = req.user._id;

    const cart = await Cart.findOne({ userId: userId });

    // Check user has a cart or no
    if (!cart) {
      return res.status(404).json({
        success: false,
        message:
          "Cart not found. Please add products to your cart before ordering.",
      });
    }

    // Check if user has a product in his cart or no

    if (cart.items.length === 0) {
      return res.status(400).json({
        success: false,
        message:
          "Cart  found. Please add products to your cart before ordering.",
      });
    }

    // i come here if user has a cart and then there are products in cart

    //Step 2:

    // fetch all productId in an array there can be mulitple so using array
    const allProductsId = cart.items.map((items) => {
      return items.productId;
    });

    // check those allProductId exist in the product collection all should exist

    // design decision if user has 3 products 2 exist 1 does not then fail entire cart order

    // checks the product id and fetches all matching

    // if none matches allProducts =[] if one matches allProducts = [{document 1 with product}]
    const allProducts = await Product.find({
      _id: { $in: allProductsId },
    });

    // SO make sure allProducts exists

    if (allProducts.length !== cart.items.length) {
      return res.status(400).json({
        success: false,
        message:
          "One or more products in your cart no longer exist. Please update your cart.",
      });
    }

    // validate qty logic
    for (const item of cart.items) {
      if (!Number.isInteger(item.quantity) || item.quantity <= 0) {
        return res.status(400).json({
          success: false,
          message: "Invalid product quantity in cart.",
        });
      }
    }

    // orderItems you have an array of objects of each product with latest price

    const orderItems = cart.items.map((item) => {
      const product = allProducts.find(
        (product) => product._id.toString() === item.productId.toString(),
      );

      return {
        productId: product._id,
        price: product.price,
        quantity: item.quantity,
      };
    });

    // Step 3 Calculate total Amount

    const calculateAmount = orderItems.reduce((acc, item) => {
      return acc + item.price * item.quantity;
    }, 0);

    // Step 4: Create the order
    const order = await Order.create({
      userId,
      items: orderItems,
      totalAmount: calculateAmount,
      shippingAddress: req.body.shippingAddress,
    });

    // Step 6: Clear the cart only after order creation succeeds
    cart.items = [];
    await cart.save();

    // Step 7: Return the response
    return res.status(201).json({
      success: true,
      message: "Order placed successfully",
      data: {
        orderId: order._id,
        items: order.items,
        totalAmount: order.totalAmount,
        status: order.status,
        paymentStatus: order.paymentStatus,
        shippingAddress: order.shippingAddress,
        createdAt: order.createdAt,
      },
    });
  } catch (err) {
    return res.status(500).json({
      message: "Internal Server Error",
    });
  }

  /*
* If user is authenticated you will come to this createOrder fn
* Step 0 : validate the requestBody hah shipping address
* Step 1 
        * Fetch the userId from req.user._id
        * check if the user owns the cart or no 
                * if no then return please  create cart before ordering empty cart  with no products cannot be ordered
                * if user has a cart but products are empty then please add products to cart
                * if user has a cart and products in it go to step2
* Step 2 : Validate the products in the cart
        *   Does every product in the cart still exist in the product collection

export const cancelOrderById = async (req, res) => {
  try {
    const orderId = req.params.orderId;

    // Step 1: Validate MongoDB ObjectId
    if (!mongoose.isValidObjectId(orderId)) {
      return res.status(400).json({
        message: "Invalid MongoDB ObjectId",
      });
    }

    // Step 2: Find order belonging to logged-in user
    const order = await Order.findOne({
      _id: orderId,
      userId: req.user._id,
    });

    if (!order) {
      return res.status(404).json({
        message: "No such order exists",
      });
    }

    // Step 3: Check whether cancellation is allowed
    if (order.status !== "pending") {
      return res.status(400).json({
        message: "Order cannot be cancelled",
      });
    }

    // Step 4: Update order status
    order.status = "cancelled";
    await order.save();

    // Step 5: Success response

import jwt from "jsonwebtoken";
import User from "../models/user.js";

export const userAuth = async (req, res, next) => {

  console.log("User Auth");
  // Step 1: Check whether the browser sent the authentication cookie
  const { token } = req.cookies ?? {};

  if (!token) {
    return res.status(401).json({
      message: "Please login first",
    });
  }

  // Step 2: Verify the JWT separately
  // If the token is invalid, malformed, or expired, return 401
  let decoded;

  try {
    decoded = jwt.verify(token, process.env.JWT_SECRET);
  } catch (err) {
    return res.status(401).json({
      message: "Invalid or expired token",
    });
  }

  // Step 3: Extract the user ID from the verified JWT payload
  // Your login controller creates the token using { id: user._id }
  const { id } = decoded;

  // Step 4: Find the user in MongoDB
  // Database errors should not be treated as invalid-token errors
  try {
    const user = await User.findById(id).select("-password");

    // Step 5: If the user no longer exists, reject the request
    if (!user) {
      return res.status(401).json({
        message: "User not found",
      });
    }

    // Step 6: Attach the authenticated user to the request
    // Later middleware and controllers can access req.user
    req.user = user;

    // Step 7: Continue to the next middleware or route handler
    return next();
  } catch (err) {
    // Unexpected database or server error
    return res.status(500).json({
      message: "Internal server error",
    });
  }
};

export const validateProduct = (reqBody) => {
  // Step 1: Check that the request body is a valid object
  if (!reqBody || typeof reqBody !== "object" || Array.isArray(reqBody)) {
    return "Invalid product data";
  }

  // Step 2: Reject fields that are not allowed
  const allowedFields = [
    "name",
    "description",
    "price",
    "stock",
    "category",
    "image",
  ];

  const requestedFields = Object.keys(reqBody);

  if (!requestedFields.every((field) => allowedFields.includes(field))) {
    return "Invalid product fields provided";
  }

  // Step 3: Extract the approved fields
  const { name, description, price, stock, category, image } = reqBody;

  // Step 4: Validate each field
  if (!name || typeof name !== "string" || name.trim().length === 0) {
    return "Product name is required";
  }

  if (!description || typeof description !== "string") {
    return "Product description is required";
  }

  if (typeof price !== "number" || price < 0 || !Number.isFinite(price)) {
    return "Price must be a valid non-negative number";
  }

  if (!Number.isInteger(stock) || stock < 0) {
    return "Stock must be a non-negative integer";
  }

  if (
    !category ||
    typeof category !== "string" ||
    category.trim().length === 0
  ) {
    return "Product category is required";
  }

  if (!image || typeof image !== "object" || Array.isArray(image)) {
    return "Product image is required";
  }

  if (
    typeof image.url !== "string" ||
    image.url.trim().length === 0 ||
    typeof image.publicId !== "string" ||
    image.publicId.trim().length === 0
  ) {
    return "Image must contain a valid url and publicId";
  }

  return null;
};

import validator from "validator";

export const validateUser = (reqBody) => {
  // Step 1: Check the user fields can send only the required fields
  const allowedFields = ["name", "email", "password"];

  // Check if reqBody is a valid object

  if (!reqBody || typeof reqBody !== "object" || Array.isArray(reqBody)) {
    return "Invalid request body";
  }

  const requestedFields = Object.keys(reqBody);

  const isAllowed = requestedFields.every((field) =>
    allowedFields.includes(field),
  );

  if (!isAllowed) {
    return "Invalid fields provided";
  }

  const { name, email, password } = reqBody;

  if (typeof name !== "string" || !name.trim()) {
    return "Name is required";
  }

  if (typeof email !== "string" || !email.trim()) {
    return "Email is required";
  }

  if (typeof password !== "string" || !password) {
    return "Password is required";
  }

  //Step 2 : Check name should not be empty

  if (name.trim().length < 3) {
    return "Name must be at least 3 characters";
  }

  // Validate email through validate npm package

  if (!validator.isEmail(email)) {
    // if it passes it returns true so if not passes it returns false
    return "Invalid email";
  }

  // Validate strong password weak password gets rejected

  if (!validator.isStrongPassword(password)) {
    return "Password is too weak";
  }

  return null; // means all the checks are passed
};
